cmake_minimum_required(VERSION 3.20)
project(ez3fs VERSION 0.24.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(FUSE3 QUIET IMPORTED_TARGET fuse3)
    pkg_check_modules(LIBUSB QUIET IMPORTED_TARGET libusb-1.0)
endif()

set(EZ3FS_SOURCES src/ez3fs_archive.cpp src/ez3fs_archive_comparison.cpp src/ez3fs_archive_loader.cpp src/ez3fs_live_filesystem.cpp src/ez3fs_new_image_file.cpp src/ez3fs_recovery_snapshot.cpp src/ez3fs_virtual_filesystem.cpp)
add_executable(ez3fs ez3fs.cpp src/ez3fs_fuse_mount.cpp src/ez3fs_cartridge_storage.cpp src/ez3fs_cartridge_live_device.cpp ${EZ3FS_SOURCES})
target_include_directories(ez3fs PRIVATE include)
if(TARGET PkgConfig::FUSE3)
    target_compile_definitions(ez3fs PRIVATE EZ3FS_HAS_FUSE3=1)
    target_link_libraries(ez3fs PRIVATE PkgConfig::FUSE3)
endif()
if(TARGET PkgConfig::LIBUSB)
    target_compile_definitions(ez3fs PRIVATE EZ3FS_HAS_LIBUSB=1)
    target_link_libraries(ez3fs PRIVATE PkgConfig::LIBUSB)
endif()
if(MSVC)
    target_compile_options(ez3fs PRIVATE /W4 /permissive-)
else()
    target_compile_options(ez3fs PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    function(ez3fs_add_test name)
        add_executable(${name} ${ARGN} ${EZ3FS_SOURCES})
        target_include_directories(${name} PRIVATE include)
        add_test(NAME ${name} COMMAND ${name})
    endfunction()
    ez3fs_add_test(ez3fs_archive_test tests/ez3fs_archive_test.cpp)
    ez3fs_add_test(ez3fs_virtual_filesystem_test tests/ez3fs_virtual_filesystem_test.cpp)
    ez3fs_add_test(ez3fs_new_image_file_test tests/ez3fs_new_image_file_test.cpp)
    ez3fs_add_test(ez3fs_archive_comparison_test tests/ez3fs_archive_comparison_test.cpp)
    ez3fs_add_test(ez3fs_recovery_snapshot_test tests/ez3fs_recovery_snapshot_test.cpp)
    ez3fs_add_test(ez3fs_live_filesystem_test tests/ez3fs_live_filesystem_test.cpp)
endif()
