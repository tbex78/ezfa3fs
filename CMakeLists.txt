cmake_minimum_required(VERSION 3.20)
project(ez3fs VERSION 0.40.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(FUSE3 QUIET IMPORTED_TARGET fuse3)
    pkg_check_modules(LIBUSB QUIET IMPORTED_TARGET libusb-1.0)
endif()

set(EZ3FS_SOURCES src/ez3fs_archive.cpp src/ez3fs_archive_comparison.cpp src/ez3fs_archive_loader.cpp src/ez3fs_live_filesystem.cpp src/ez3fs_live_mount_backend.cpp src/ez3fs_mount_session.cpp src/ez3fs_new_image_file.cpp src/ez3fs_recovery_snapshot.cpp src/ez3fs_virtual_filesystem.cpp src/ez3fs_virtual_mount_backend.cpp)
set(EZ3FS_CLI_SOURCES ez3fs.cpp src/ez3fs_fuse_mount.cpp src/ez3fs_cartridge_storage.cpp src/ez3fs_cartridge_live_device.cpp src/ez3fs_live_cartridge_session.cpp ${EZ3FS_SOURCES})

function(ez3fs_configure_cli target)
    target_include_directories(${target} PRIVATE include)
    if(TARGET PkgConfig::FUSE3)
        target_compile_definitions(${target} PRIVATE EZ3FS_HAS_FUSE3=1)
        target_link_libraries(${target} PRIVATE PkgConfig::FUSE3)
    endif()
    if(TARGET PkgConfig::LIBUSB)
        target_compile_definitions(${target} PRIVATE EZ3FS_HAS_LIBUSB=1)
        target_link_libraries(${target} PRIVATE PkgConfig::LIBUSB)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_executable(ez3fs ${EZ3FS_CLI_SOURCES})
ez3fs_configure_cli(ez3fs)

add_executable(ezfs-legacy ${EZ3FS_CLI_SOURCES})
target_compile_definitions(ezfs-legacy PRIVATE EZ3FS_LEGACY_CLI=1)
ez3fs_configure_cli(ezfs-legacy)

include(CTest)
if(BUILD_TESTING)
    function(ez3fs_add_test name)
        add_executable(${name} ${ARGN} ${EZ3FS_SOURCES})
        target_include_directories(${name} PRIVATE include)
        add_test(NAME ${name} COMMAND ${name})
    endfunction()
    ez3fs_add_test(ez3fs_archive_test tests/ez3fs_archive_test.cpp)
    ez3fs_add_test(ez3fs_virtual_filesystem_test tests/ez3fs_virtual_filesystem_test.cpp)
    ez3fs_add_test(ez3fs_new_image_file_test tests/ez3fs_new_image_file_test.cpp)
    ez3fs_add_test(ez3fs_archive_comparison_test tests/ez3fs_archive_comparison_test.cpp)
    ez3fs_add_test(ez3fs_recovery_snapshot_test tests/ez3fs_recovery_snapshot_test.cpp)
    ez3fs_add_test(ez3fs_live_filesystem_test tests/ez3fs_live_filesystem_test.cpp)
    ez3fs_add_test(ez3fs_live_mount_backend_test tests/ez3fs_live_mount_backend_test.cpp)
    ez3fs_add_test(ez3fs_mount_session_test tests/ez3fs_mount_session_test.cpp)
    add_test(NAME ez3fs_live_cli_version COMMAND ez3fs --version)
    set_tests_properties(ez3fs_live_cli_version PROPERTIES PASS_REGULAR_EXPRESSION "ez3fs 0.40.1")
    add_test(NAME ezfs_legacy_cli_version COMMAND ezfs-legacy --version)
    set_tests_properties(ezfs_legacy_cli_version PROPERTIES PASS_REGULAR_EXPRESSION "ezfs-legacy 0.40.1")
endif()
